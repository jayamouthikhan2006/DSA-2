def count_sort(a):
    if any(x<0 for x in a): raise ValueError("Negative input not allowed")
    c=[0]*(max(a,default=0)+1)
    for x in a:c[x]+=1
    return [i for i,n in enumerate(c) for _ in range(n)]

9. Infix to Postfix

def post(e):
    p={'+':1,'-':1,'*':2,'/':2,'^':3};s=[];o=''
    for c in e:
        if c.isalnum():o+=c
        elif c=='(':s.append(c)
        elif c==')':
            while s[-1]!='(':o+=s.pop()
            s.pop()
        else:
            while s and s[-1]!='(' and (p[s[-1]]>p[c] or p[s[-1]]==p[c] and c!='^'):o+=s.pop()
            s.append(c)
    return o+''.join(s[::-1])

10. Balanced Parentheses

def bal(e):
    s=[];m={')':'(','}':'{',']':'['}
    for c in e:
        if c in m.values():s.append(c)
        elif c in m and (not s or s.pop()!=m[c]):return False
    return not s

25. Cycle Detection (Directed Graph)

def cyc(g):
    vis=set();st=set()
    def d(u):
        vis.add(u);st.add(u)
        for v in g[u]:
            if v in st or v not in vis and d(v):return True
        st.discard(u);return False
    return any(d(u) for u in g if u not in vis)

Output
